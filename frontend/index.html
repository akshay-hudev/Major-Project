Python Integration Example

import requests

url = "http://127.0.0.1:8000/api/predict"

# Fetch a ready-to-test patient trajectory
scenario = requests.get("http://127.0.0.1:8000/api/scenarios/DEMO-002").json()
payload = scenario["sample_data"]

# Run temporal inference
response = requests.post(url, json=payload)
result = response.json()

print(f"Risk Probability: {result['probability_percent']}")
print(f"Classification:   {result['classification']}")
print(f"Risk Tier:        {result['risk_level']}")
for driver in result["top_contributors"][:3]:
    print(f" - {driver['feature_name']}: {driver['temporal_descriptor']}")
